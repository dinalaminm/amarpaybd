💳 AmarPayBd — Client Integration Guide

ক্লায়েন্ট শুধু License Key এবং API URL পাবে। Firebase config বা কোনো credentials লাগবে না।

1. JavaScript / Web Integration

// তোমার website/app এ এভাবে call করবে

async function verifyPayment(trxId, amount, method) {

  const response = await fetch('https://yourdomain.com/api/verify', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      licenseKey: 'AMARPAY-AXKP-QMRV-BNJT',  // তোমার license key
      trxId:      trxId,                          // customer এর দেওয়া TrxID
      amount:     amount,                         // expected amount (optional)
      method:     method                          // "bKash" / "Nagad" / "Rocket"
    })
  });

  const data = await response.json();

  if (data.verified) {
    // ✅ Payment সফল
    console.log('Payment verified!', data.amount, data.sender);
    // এখানে order confirm করো
  } else {
    // ❌ Payment ব্যর্থ
    console.log('Failed:', data.reason);
    // error দেখাও
  }
}

// ব্যবহার:
verifyPayment('8H7K2ABC1', 500, 'bKash');

2. PHP Integration

<?php
function verifyPayment($trxId, $amount, $method) {
    $data = [
        'licenseKey' => 'AMARPAY-AXKP-QMRV-BNJT',
        'trxId'      => $trxId,
        'amount'     => $amount,
        'method'     => $method
    ];

    $ch = curl_init('https://yourdomain.com/api/verify');
    curl_setopt($ch, CURLOPT_POST, true);
    curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($data));
    curl_setopt($ch, CURLOPT_HTTPHEADER, ['Content-Type: application/json']);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);

    $response = json_decode(curl_exec($ch), true);
    curl_close($ch);

    return $response;
}

// ব্যবহার:
$result = verifyPayment('8H7K2ABC1', 500, 'bKash');

if ($result['verified']) {
    // ✅ Payment confirmed — order process করো
    echo "Payment verified: " . $result['amount'] . " BDT";
} else {
    // ❌ Failed
    echo "Failed: " . $result['reason'];
}
?>

3. API Response Format

// ✅ Success Response:
{
  "verified":  true,
  "trxId":     "8H7K2ABC1",
  "amount":    500.0,
  "method":    "bKash",
  "sender":    "01711xxxxxx",
  "timestamp": "2024-01-15 10:30:00"
}

// ❌ Fail Response:
{
  "verified": false,
  "reason":   "Transaction not found"
  // reason এর সম্ভাব্য মান:
  // "Invalid license key"
  // "License key is inactive"
  // "License key has expired"
  // "Transaction not found"
  // "Transaction already used"
  // "Amount mismatch"
  // "Payment method mismatch"
}
